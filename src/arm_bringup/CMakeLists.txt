cmake_minimum_required(VERSION 3.20)
project(arm_bringup)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY launch worlds DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE)
install(PROGRAMS scripts/check_stack.py DESTINATION lib/${PROJECT_NAME} RENAME check_stack)
install(PROGRAMS scripts/load_scene.py DESTINATION lib/${PROJECT_NAME} RENAME load_scene)
install(PROGRAMS scripts/workcell_ui.py DESTINATION lib/${PROJECT_NAME} RENAME workcell_ui)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(scene_geometry test/test_scene.py)
endif()
ament_package()
